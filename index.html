4. 雲端資料庫 Firestore Rules (安全性規則設定指引)

為什麼名單沒存進雲端？若您的 Firebase 專案尚未設定讀取/寫入規則，或未開啟「Authentication → Sign-in method → 匿名 (Anonymous) 與 Email/Password」，Firestore 會拒絕寫入。請至 Firebase Console → Firestore Database → 規則 (Rules) 貼上以下規則並點擊「發布」： 

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // 開放 OmniDraw 活動名單、搖珠群組與 SaaS 會員資料讀寫
    match /omnidraw_events/{eventId} {
      allow read, write: if true;
    }
    match /saas_members/{memberId} {
      allow read, write: if true;
    }
    match /artifacts/{appId}/public/data/{document=**} {
      allow read, write: if true;
    }
  }
}
