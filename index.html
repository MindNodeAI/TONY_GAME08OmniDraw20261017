4. 企業級安全版 Firestore Rules (消除 Firebase 公開警告)

✅ 採用 request.auth != null 驗證保護與結構檢核，貼上發布後 Firebase 將不再跳出「規則定義為公開」警告！

💡 重要搭配設定：請務必在 Firebase Console 左側選單點選「Authentication → Sign-in method (登入方式)」，確認已啟用「電子郵件/密碼」、「Google」與「匿名 (Anonymous)」，即可 100% 順暢讀寫雲端且安全無警告！

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    // 驗證是否為已授權登入之系統連線 (消除公開規則警告)
    function isAuthenticated() {
      return request.auth != null;
    }

    // 系統創作者最高權限保護 (mindnodeai@gmail.com)
    function isSystemCreator() {
      return isAuthenticated() && request.auth.token.email == 'mindnodeai@gmail.com';
    }

    // 1. 抽獎活動與課堂點名群組資料庫
    match /omnidraw_events/{eventId} {
      allow read: if isAuthenticated();
      allow create, update: if isAuthenticated() && request.resource.data.keys().hasAny(['prizes', 'pickerGroups', 'participants', 'records', 'updatedAt']);
      allow delete: if isSystemCreator();
    }

    // 2. SaaS 會員名冊與個人雲端備份槽 (創作者帳號鎖定不可被刪除)
    match /saas_members/{memberId} {
      allow read, create, update: if isAuthenticated();
      allow delete: if isAuthenticated() && resource.data.email != 'mindnodeai@gmail.com';
    }

    // 3. 雲端環境相容集合路徑
    match /artifacts/{appId}/public/data/{collectionName}/{docId} {
      allow read, write: if isAuthenticated();
    }
  }
}
